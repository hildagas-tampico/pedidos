/* * Hilda Gas — registro de pedidos por mes. * * Conserva la pestaña histórica "Pedidos". * Nuevas pestañas: Pedidos-AAAA-MM. * Nuevos folios: HG-AAAAMM-0001. * * La hoja de cálculo debe permanecer privada. * El mes se determina con la zona horaria America/Mexico_City. * * Límites comerciales para relleno por litros o pesos * en cilindro portátil: * 20 kg: 40 litros; 30 kg: 55 litros; 45 kg: 90 litros. * * No aplican a intercambio, tanque estacionario ni "Lleno". * El suministro real sigue sujeto a verificación de seguridad. * * COPIA PARA PRUEBAS: * Devuelve fechaPrecios en pedidos nuevos y reintentos. * El destino depende de la propiedad HOJA_ID. * Verificar que corresponda a la hoja de pruebas antes de ejecutar. */ const CONFIG = { hoja: "Pedidos", zonaHoraria: "America/Mexico_City", preciosUrl: "https://hildagas-tampico.github.io/pedidos/precios.json" }; const LIMITES_LITROS_CILINDRO = Object.freeze({ 20: 40, 30: 55, 45: 90 }); const ENCABEZADOS = [ "Fecha de registro", "Folio", "ID de solicitud", "Servicio", "Recipiente", "Modalidad", "Capacidad kg", "Cantidad de cilindros", "Valor solicitado", "Litros estimados", "Importe estimado MXN", "Nombre", "Apellido", "Teléfono", "Calle y número", "Colonia", "Municipio", "Entre calles", "Referencias", "Consentimiento", "Fecha de precios", "Estado" ]; /* * Ejecutar manualmente solo después de comprobar el archivo de destino. * Guarda el ID, verifica precios y prepara la pestaña actual. * No borra ni mueve los pedidos existentes. * No genera ningún pedido ni consume un folio. */ function prepararHoja() { const libro = SpreadsheetApp.getActiveSpreadsheet(); if (!libro) { throw new Error( "Abre Apps Script desde Extensiones en tu hoja de cálculo." ); } const precios = cargarPrecios_(); const bloqueo = LockService.getScriptLock(); if (!bloqueo.tryLock(25000)) { throw new Error( "El registro está ocupado. Reintenta en unos segundos." ); } try { const propiedades = PropertiesService.getScriptProperties(); const idAnterior = propiedades.getProperty("HOJA_ID"); if (idAnterior && idAnterior !== libro.getId()) { throw new Error( "Este proyecto está configurado para otro archivo. " + "No cambies el archivo de destino sin revisar la configuración." ); } propiedades.setProperty("HOJA_ID", libro.getId()); // Afecta la presentación de las fechas de todo el archivo. libro.setSpreadsheetTimeZone(CONFIG.zonaHoraria); // Comprueba la pestaña histórica, si existe; no la modifica. const historica = libro.getSheetByName(CONFIG.hoja); if (historica && historica.getLastRow() > 0) { comprobarEncabezados_(historica); } const hoja = obtenerHojaMensual_(libro, new Date()); SpreadsheetApp.flush(); console.log( "Configuración lista. Pestaña actual: " + hoja.getName() + ". Fecha de precios: " + precios.date ); } finally { bloqueo.releaseLock(); } } /* * Comprueba que la aplicación web responde. * No muestra datos personales. */ function doGet() { return responder_({ ok: true, mensaje: "Registro Hilda Gas disponible", version: "pedidos-mensuales-v1" }); } /* * Recibe el pedido enviado por el catálogo. */ function doPost(e) { let bloqueo; let bloqueado = false; try { if (!e || !e.postData || !e.postData.contents) { throw new Error("No se recibió un pedido."); } if (e.postData.contents.length > 15000) { throw new Error("La solicitud es demasiado grande."); } let pedido; try { pedido = JSON.parse(e.postData.contents); } catch (_) { throw new Error("El formato del pedido no es válido."); } const validado = validarPedido_(pedido); bloqueo = LockService.getScriptLock(); if (!bloqueo.tryLock(25000)) { throw new Error( "El registro está ocupado. Reintenta en unos segundos." ); } bloqueado = true; const libro = obtenerLibro_(); /* * Busca también en meses anteriores y en "Pedidos". * Si la respuesta se perdió y se reintenta con el mismo ID, * devuelve el registro original sin crear otro pedido. */ const existente = buscarSolicitud_(libro, validado.requestId); if (existente) { /* * Conserva la fecha de precios del pedido original. * Sheets puede devolver la celda como Date o como texto. */ const fechaPreciosOriginal = existente[20] instanceof Date ? Utilities.formatDate( existente[20], CONFIG.zonaHoraria, "yyyy-MM-dd" ) : String(existente[20] || ""); return responder_({ ok: true, folio: String(existente[1]), litros: existente[9] === "" ? null : Number(existente[9]), total: existente[10] === "" ? null : Number(existente[10]), fechaPrecios: fechaPreciosOriginal }); } const precios = cargarPrecios_(); const cotizacion = cotizar_(validado, precios); /* * Usa una sola fecha para elegir pestaña, folio y registro, * incluso si el proceso coincide con un cambio de mes. */ const fechaRegistro = new Date(); const hoja = obtenerHojaMensual_(libro, fechaRegistro); const folio = siguienteFolio_(hoja, fechaRegistro); const d = validado.datos; hoja.appendRow([ fechaRegistro, folio, validado.requestId, validado.servicio, validado.recipiente, validado.modalidad, validado.capacidad || "", validado.cantidad, validado.valor || "", cotizacion.litros === null ? "" : cotizacion.litros, cotizacion.total === null ? "" : cotizacion.total, textoSeguro_(d.nombre), textoSeguro_(d.apellido), textoSeguro_(d.telefono), textoSeguro_(d.calle), textoSeguro_(d.colonia), textoSeguro_(d.municipio), textoSeguro_(d.entreCalles), textoSeguro_(d.referencias), "Sí", precios.date, "Pendiente de confirmar" ]); SpreadsheetApp.flush(); return responder_({ ok: true, folio: folio, litros: cotizacion.litros, total: cotizacion.total, fechaPrecios: precios.date }); } catch (error) { // No incluye los datos personales del pedido en el log. const mensaje = String( error && error.message ? error.message : "No se pudo guardar el pedido." ); console.error(mensaje); return responder_({ ok: false, error: mensaje }); } finally { if (bloqueado) bloqueo.releaseLock(); } } function obtenerLibro_() { const id = PropertiesService.getScriptProperties() .getProperty("HOJA_ID"); if (!id) { throw new Error( "Falta configuración: ejecuta prepararHoja desde el editor." ); } return SpreadsheetApp.openById(id); } function obtenerHojaMensual_(libro, fecha) { const mes = Utilities.formatDate( fecha, CONFIG.zonaHoraria, "yyyy-MM" ); const nombre = CONFIG.hoja + "-" + mes; let hoja = libro.getSheetByName(nombre); if (!hoja) { hoja = libro.insertSheet(nombre); } if (hoja.getLastRow() === 0) { prepararEncabezados_(hoja); } else { comprobarEncabezados_(hoja); } return hoja; } function prepararEncabezados_(hoja) { if (hoja.getMaxColumns() < ENCABEZADOS.length) { hoja.insertColumnsAfter( hoja.getMaxColumns(), ENCABEZADOS.length - hoja.getMaxColumns() ); } hoja.getRange(1, 1, 1, ENCABEZADOS.length) .setValues([ENCABEZADOS]) .setBackground("#0c087c") .setFontColor("#ffffff") .setFontWeight("bold"); hoja.setFrozenRows(1); hoja.getRange("A:A").setNumberFormat("dd/MM/yyyy HH:mm:ss"); hoja.getRange("B:C").setNumberFormat("@"); hoja.getRange("J:K").setNumberFormat("0.00"); hoja.getRange("N:N").setNumberFormat("@"); hoja.setColumnWidth(1, 165); hoja.setColumnWidth(2, 165); hoja.setColumnWidth(3, 220); hoja.setColumnWidth(22, 180); } function comprobarEncabezados_(hoja) { if (hoja.getMaxColumns() < ENCABEZADOS.length) { throw new Error( "Faltan columnas en la pestaña " + hoja.getName() + ". No borres pedidos; revisa sus encabezados." ); } const actuales = hoja.getRange( 1, 1, 1, ENCABEZADOS.length ).getValues()[0]; if (!ENCABEZADOS.every((titulo, i) => actuales[i] === titulo)) { throw new Error( "Los encabezados de la pestaña " + hoja.getName() + " no coinciden con el registro de pedidos. " + "No borres datos ni cambies los títulos sin revisar." ); } } function esHojaDePedidos_(hoja) { const nombre = hoja.getName(); return nombre === CONFIG.hoja || /^Pedidos-\d{4}-(0[1-9]|1[0-2])$/.test(nombre); } function buscarSolicitud_(libro, requestId) { /* * Solo consulta "Pedidos" y "Pedidos-AAAA-MM". * Otras pestañas del archivo no se utilizan. */ const hojas = libro.getSheets() .filter(esHojaDePedidos_) .sort((a, b) => b.getName().localeCompare(a.getName())); for (const hoja of hojas) { if (hoja.getLastRow() === 0) continue; comprobarEncabezados_(hoja); if (hoja.getLastRow() <= 1) continue; const encontrado = hoja.getRange( 2, 3, hoja.getLastRow() - 1, 1 ).createTextFinder(requestId) .matchEntireCell(true) .useRegularExpression(false) .findNext(); if (encontrado) { return hoja.getRange( encontrado.getRow(), 1, 1, ENCABEZADOS.length ).getValues()[0]; } } return null; } function siguienteFolio_(hoja, fecha) { /* * Se ejecuta dentro del bloqueo de doPost. * Cada mes tiene su contador propio. * También revisa los folios guardados en la pestaña. */ const periodo = Utilities.formatDate( fecha, CONFIG.zonaHoraria, "yyyyMM" ); const clave = "ULTIMO_FOLIO_" + periodo; const propiedades = PropertiesService.getScriptProperties(); const guardado = Number(propiedades.getProperty(clave)); let ultimo = Number.isSafeInteger(guardado) && guardado >= 0 ? guardado : 0; const patron = new RegExp("^HG-" + periodo + "-(\\d+)$"); if (hoja.getLastRow() > 1) { const folios = hoja.getRange( 2, 2, hoja.getLastRow() - 1, 1 ).getValues(); folios.forEach(fila => { const coincidencia = patron.exec(String(fila[0])); if (coincidencia) { const numero = Number(coincidencia[1]); if (!Number.isSafeInteger(numero)) { throw new Error("Se encontró un folio fuera de rango."); } ultimo = Math.max(ultimo, numero); } }); } const siguiente = ultimo + 1; if (!Number.isSafeInteger(siguiente)) { throw new Error("El contador de folios está fuera de rango."); } /* * Se reserva antes de escribir para no reutilizarlo si hay * un fallo de escritura. Puede haber saltos en el consecutivo. */ propiedades.setProperty(clave, String(siguiente)); return "HG-" + periodo + "-" + String(siguiente).padStart(4, "0"); } function cargarPrecios_() { let respuesta; try { respuesta = UrlFetchApp.fetch(CONFIG.preciosUrl.trim(), { muteHttpExceptions: true, followRedirects: true }); } catch (_) { throw new Error( "No se pudo consultar el archivo público de precios." ); } if (respuesta.getResponseCode() !== 200) { throw new Error( "No se encontró precios.json. Revisa preciosUrl y GitHub Pages." ); } let precios; try { precios = JSON.parse(respuesta.getContentText()); } catch (_) { throw new Error("precios.json no contiene un JSON válido."); } if (!precios || typeof precios !== "object" || Array.isArray(precios)) { throw new Error("El archivo de precios no es válido."); } for (const campo of ["kg", "lt", "c20", "c30", "c45"]) { if ( typeof precios[campo] !== "number" || !Number.isFinite(precios[campo]) || precios[campo] <= 0 ) { throw new Error( "El precio " + campo + " debe ser un número positivo." ); } } if ( typeof precios.date !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(precios.date) ) { throw new Error( "La fecha de precios.json debe tener formato AAAA-MM-DD." ); } const fecha = new Date(precios.date + "T00:00:00Z"); if ( !Number.isFinite(fecha.getTime()) || fecha.toISOString().slice(0, 10) !== precios.date ) { throw new Error("La fecha del archivo de precios no es válida."); } return precios; } function validarPedido_(p) { if (!p || typeof p !== "object" || Array.isArray(p)) { throw new Error("Pedido no válido."); } if ( typeof p.requestId !== "string" || !/^[a-zA-Z0-9-]{16,80}$/.test(p.requestId) ) { throw new Error("El ID de solicitud no es válido."); } if (!["relleno", "intercambio"].includes(p.servicio)) { throw new Error("Selecciona un servicio válido."); } if (!["estacionario", "cilindro"].includes(p.recipiente)) { throw new Error("Selecciona un recipiente válido."); } if ( !p.datos || typeof p.datos !== "object" || Array.isArray(p.datos) ) { throw new Error("Faltan los datos de entrega."); } if (p.datos.website) { throw new Error("Solicitud no válida."); } if (p.consentimiento !== true) { throw new Error("Falta la autorización para registrar los datos."); } const capacidad = Number(p.capacidad); const cantidad = Number(p.cantidad); const valor = Number(p.valor); if ( p.recipiente === "cilindro" && ![20, 30, 45].includes(capacidad) ) { throw new Error("La capacidad del cilindro no es válida."); } if (p.servicio === "intercambio") { if ( p.recipiente !== "cilindro" || p.modalidad !== "intercambio" || !Number.isInteger(cantidad) || cantidad < 1 || cantidad > 10 ) { throw new Error("Los datos del intercambio no son válidos."); } } else { if (!["litros", "pesos", "lleno"].includes(p.modalidad)) { throw new Error("Selecciona una modalidad de relleno válida."); } if ( p.modalidad !== "lleno" && (!Number.isFinite(valor) || valor <= 0 || valor > 100000) ) { throw new Error("La cantidad solicitada no es válida."); } } const datos = {}; const campos = { nombre: [60, true], apellido: [60, true], telefono: [20, true], calle: [160, true], colonia: [100, true], municipio: [100, true], entreCalles: [160, false], referencias: [500, false] }; Object.keys(campos).forEach(campo => { const [maximo, obligatorio] = campos[campo]; const entrada = p.datos[campo]; if (entrada != null && typeof entrada !== "string") { throw new Error("El campo " + campo + " no es válido."); } const texto = (entrada || "").trim(); if ((obligatorio && !texto) || texto.length > maximo) { throw new Error("Revisa el campo " + campo + "."); } datos[campo] = texto; }); if (!/^(?:52)?\d{10}$/.test(datos.telefono.replace(/\D/g, ""))) { throw new Error( "El teléfono debe tener 10 dígitos o 52 y 10 dígitos." ); } return { requestId: p.requestId, servicio: p.servicio, recipiente: p.recipiente, modalidad: p.modalidad, capacidad: p.recipiente === "cilindro" ? capacidad : 0, cantidad: p.servicio === "intercambio" ? cantidad : 1, valor: p.servicio === "relleno" && p.modalidad !== "lleno" ? valor : 0, datos: datos }; } function validarLimiteCilindro_(p, litros) { if ( p.servicio !== "relleno" || p.recipiente !== "cilindro" || !["litros", "pesos"].includes(p.modalidad) ) { return; } const maximo = LIMITES_LITROS_CILINDRO[p.capacidad]; if (!Number.isFinite(maximo)) { throw new Error("La capacidad del cilindro no es válida."); } if (!Number.isFinite(litros) || litros <= 0) { throw new Error("La cantidad calculada de litros no es válida."); } // Margen mínimo para errores de aritmética decimal. if (litros > maximo + 1e-9) { throw new Error( "El límite configurado para el cilindro de " + p.capacidad + " kg es de " + maximo + " litros. Reduce los litros o el importe solicitado." ); } } function cotizar_(p, precios) { let total = null; let litros = null; if (p.servicio === "intercambio") { total = precios["c" + p.capacidad] * p.cantidad; } else if (p.modalidad === "litros") { litros = p.valor; total = p.valor * precios.lt; } else if (p.modalidad === "pesos") { total = p.valor; litros = p.valor / precios.lt; } else if (p.recipiente === "cilindro") { total = p.capacidad * precios.kg; } validarLimiteCilindro_(p, litros); return { total: total === null ? null : redondear_(total), litros: litros === null ? null : redondear_(litros) }; } function redondear_(numero) { return Math.round((numero + Number.EPSILON) * 100) / 100; } // Impide interpretar texto del cliente como fórmula en Sheets. function textoSeguro_(texto) { const valor = String(texto || ""); return /^[=+\-@\t\r\n]/.test(valor) ? "'" + valor : valor; } function responder_(objeto) { return ContentService .createTextOutput(JSON.stringify(objeto)) .setMimeType(ContentService.MimeType.JSON); }
